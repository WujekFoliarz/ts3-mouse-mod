cmake_minimum_required(VERSION 3.20)
project(TS3Mouse LANGUAGES CXX)

option(TS3_BUILD_TESTS "Build portable input regression tests" OFF)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Toy Story 3 Complete Edition needs a 64-bit DLL. Use -A x64 with Visual Studio.")
  endif()
  add_library(dinput8 SHARED src/TS3Mouse.cpp src/dinput8.def)
  target_compile_features(dinput8 PRIVATE cxx_std_17)
  target_compile_definitions(dinput8 PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601)
  target_link_libraries(dinput8 PRIVATE bcrypt user32)
  set_target_properties(dinput8 PROPERTIES PREFIX "" OUTPUT_NAME "dinput8")
  if(MSVC)
    target_compile_options(dinput8 PRIVATE /W4 /permissive- /EHsc)
    set_property(TARGET dinput8 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(dinput8 PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_options(dinput8 PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  add_custom_command(TARGET dinput8 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/TS3Mouse.ini" "$<TARGET_FILE_DIR:dinput8>/TS3Mouse.ini")
elseif(NOT TS3_BUILD_TESTS)
  message(FATAL_ERROR "Use -DCMAKE_TOOLCHAIN_FILE=cmake/mingw64.cmake to build the Windows mod on Linux.")
endif()

if(TS3_BUILD_TESTS)
  enable_testing()
  add_executable(input_tests tests/input_tests.cpp)
  target_compile_features(input_tests PRIVATE cxx_std_17)
  add_test(NAME input_regressions COMMAND input_tests)
  if(WIN32)
    add_executable(hook_harness tests/hook_harness.cpp)
    target_compile_features(hook_harness PRIVATE cxx_std_17)
    target_compile_definitions(hook_harness PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE _WIN32_WINNT=0x0601 DIRECTINPUT_VERSION=0x0800)
    target_link_libraries(hook_harness PRIVATE bcrypt user32 dxguid)
    if(MINGW)
      target_link_options(hook_harness PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
    add_test(NAME hook_abi COMMAND hook_harness)
  endif()
endif()
